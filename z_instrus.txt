# Test para Control Hazards: beq, bne, jal, jalr
# Verificar flushing del pipeline

# Test 1: beq tomado
addi x1, x0, 5      # x1 = 5
addi x2, x0, 5      # x2 = 5  
beq x1, x2, target1 # Si x1 == x2, saltar (TOMADO)
addi x3, x0, 99     # Esta instrucción debe ser flushed
addi x4, x0, 99     # Esta instrucción debe ser flushed

target1:
addi x5, x0, 10     # x5 = 10 (debe ejecutarse después del salto)

# Test 2: beq no tomado  
addi x6, x0, 1      # x6 = 1
addi x7, x0, 2      # x7 = 2
beq x6, x7, target2 # Si x6 == x7, saltar (NO TOMADO)
addi x8, x0, 20     # x8 = 20 (debe ejecutarse, no flush)

target2:
addi x9, x0, 30     # x9 = 30

# Test 3: jal
jal x10, target3    # x10 = PC+4, saltar a target3
addi x11, x0, 99    # Esta debe ser flushed
addi x12, x0, 99    # Esta debe ser flushed

target3:
addi x13, x0, 40    # x13 = 40

nop
